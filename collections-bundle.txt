const SCENE_KEYS = [
    { p: 0.0,  x: 4.4,  y: 0.1,  z: 0,   s: 1.0 },
    { p: 0.16, x: 4.4,  y: 0.1,  z: 0,   s: 1.0 },
    { p: 0.30, x: -4.6, y: -0.3, z: -2,  s: 0.9 },
    { p: 0.46, x: 0,    y: 0,    z: -10, s: 0.9 },
    { p: 0.70, x: 4.8,  y: -0.2, z: -4,  s: 0.85 },
    { p: 1.0,  x: 0,    y: 1.4,  z: -7,  s: 0.8 }

function smoothstep(t) {
    return t * t * (3 - 2 * t);
}

function sampleKeys(progress) {

    const p = Math.min(1, Math.max(0, progress));

    for (let i = 0; i < SCENE_KEYS.length - 1; i++) {

        const a = SCENE_KEYS[i];
        const b = SCENE_KEYS[i + 1];

        if (p >= a.p && p <= b.p) {

            const span = b.p - a.p;
            const t = span === 0 ? 0 : smoothstep((p - a.p) / span);

            return {
                x: a.x + (b.x - a.x) * t,
                y: a.y + (b.y - a.y) * t,
                z: a.z + (b.z - a.z) * t,
                s: a.s + (b.s - a.s) * t
            };

        }

    }

    const last = SCENE_KEYS[SCENE_KEYS.length - 1];

    return { x: last.x, y: last.y, z: last.z, s: last.s };

}

const RIM_VERTEX = `
varying vec3 vNormal;
void main() {
    vNormal = normalize(normalMatrix * normal);
    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
}`;

const RIM_FRAGMENT = `
uniform vec3 uColor;
uniform vec3 uLight;
varying vec3 vNormal;
void main() {
    vec3 n = normalize(vNormal);
    float rim = pow(1.0 - abs(n.z), 2.2);
    float lit = smoothstep(-0.15, 0.85, dot(n, normalize(uLight)));
    float a = rim * lit;
    gl_FragColor = vec4(uColor * a * 1.8, a);
}`;

const HALO_FRAGMENT = `
uniform vec3 uColor;
uniform vec3 uLight;
varying vec3 vNormal;
void main() {
    vec3 n = normalize(vNormal);
    float h = clamp(abs(n.z) / 0.55, 0.0, 1.0);
    float halo = pow(h, 2.0);
    float lit = 0.3 + 0.7 * smoothstep(-0.3, 0.9, dot(n, normalize(uLight)));
    float a = halo * lit;
    gl_FragColor = vec4(uColor * a, a);
}`;

function makeTexture(THREE, stops) {

    const size = 128;

    const canvas = document.createElement('canvas');
    canvas.width = size;
    canvas.height = size;

    const ctx = canvas.getContext('2d');

    const gradient = ctx.createRadialGradient(
        size / 2, size / 2, 0,
        size / 2, size / 2, size / 2
    );

    stops.forEach(([offset, color]) => {
        gradient.addColorStop(offset, color);
    });

    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, size, size);

    return new THREE.CanvasTexture(canvas);

}


function buildStars(THREE, track, count, rMin, rMax, size, texture, goldRatio) {

    const positions = new Float32Array(count * 3);
    const colors = new Float32Array(count * 3);

    const ivory = new THREE.Color(IVORY);
    const gold = new THREE.Color(GOLD);
    const violet = new THREE.Color(0x8a78c8);

    for (let i = 0; i < count; i++) {

        const r = rMin + Math.random() * (rMax - rMin);
        const theta = Math.random() * Math.PI * 2;
        const phi = Math.acos(2 * Math.random() - 1);

        positions[i * 3] = r * Math.sin(phi) * Math.cos(theta);
        positions[i * 3 + 1] = r * Math.sin(phi) * Math.sin(theta);
        positions[i * 3 + 2] = r * Math.cos(phi);

        const roll = Math.random();

        const base =
            roll < goldRatio
                ? gold
                : roll < goldRatio + 0.12
                    ? violet
                    : ivory;

        const k = 0.45 + Math.random() * 0.55;

        colors[i * 3] = base.r * k;
        colors[i * 3 + 1] = base.g * k;
        colors[i * 3 + 2] = base.b * k;

    }

    const geometry = track(new THREE.BufferGeometry());

    geometry.setAttribute(
        'position',
        new THREE.BufferAttribute(positions, 3)
    );

    geometry.setAttribute(
        'color',
        new THREE.BufferAttribute(colors, 3)
    );

    const material = track(new THREE.PointsMaterial({
        size,
        map: texture,
        vertexColors: true,
        transparent: true,
        opacity: 0.95,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
        sizeAttenuation: true
    }));

    return new THREE.Points(geometry, material);

}

export default class UjjayiniAboutUs extends LightningElement {

    codePillars = CODE_PILLARS;

    _initialized = false;

    _revealObserver;

    // scene state
    _ctx = null;
    _page = null;
    _layer = null;
    _running = false;
    _rafId = 0;
    _pageVisible = true;
    _disposed = false;
    _mobile = false;
    _reduceMotion = false;
    _progTarget = 0;
    _prog = 0;
    _cur = { x: 4.4, y: 0.1, z: 0, s: 1 };
    _mouse = { x: 0, y: 0 };
    _pageObserver;

    _onMouseMove = event => {

        this._mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
        this._mouse.y = (event.clientY / window.innerHeight) * 2 - 1;

    };

    _onResize = () => {

        this._resize();

        if (this._reduceMotion) {
            this._frame(0);
        }

    };

    _onVisibility = () => {

        if (document.hidden) {
            this._stop();
        } else if (this._pageVisible) {
            this._start();
        }

    };

    _onScrollStatic = () => {

        if (this._reduceMotion && this._ctx) {
            this._frame(0);
        }

    };

    _tick = now => {

        if (!this._running) {
            return;
        }

        this._rafId = window.requestAnimationFrame(this._tick);

        this._frame(now / 1000);

    };


    // =========================================================
    // LIFECYCLE
    // =========================================================

    renderedCallback() {

        if (this._initialized) {
            return;
        }

        const page =
            this.template.querySelector('.about-page');

        if (!page) {
            return;
        }

        this._initialized = true;
        this._page = page;

        this._reduceMotion =
            !!(
                window.matchMedia &&
                window.matchMedia(
                    '(prefers-reduced-motion: reduce)'
                ).matches
            );

        this._initReveal(page);

        this._initScene(page);

    }


    disconnectedCallback() {

        this._disposed = true;

        if (this._revealObserver) {
            this._revealObserver.disconnect();
            this._revealObserver = null;
        }

        this._disposeScene();

    }


    // =========================================================
    // SCROLL REVEAL
    // =========================================================

    _initReveal(page) {

        const items =
            this.template.querySelectorAll('.reveal');

        if (
            this._reduceMotion ||
            typeof IntersectionObserver === 'undefined'
        ) {

            items.forEach(item => {
                item.classList.add('is-visible');
            });

            return;

        }

        page.classList.add('is-ready');

        this._revealObserver = new IntersectionObserver(

            (entries, observer) => {

                entries.forEach(entry => {

                    if (entry.isIntersecting) {

                        entry.target.classList.add('is-visible');

                        observer.unobserve(entry.target);

                    }

                });

            },

            {
                threshold: 0.12,
                rootMargin: '0px 0px -6% 0px'
            }

        );

        items.forEach(item => {
            this._revealObserver.observe(item);
        });

    }


    // =========================================================
    // 3D SCENE
    // =========================================================

    async _initScene(page) {

        try {

            await loadScript(this, THREE_SCRIPT);

            if (this._disposed) {
                return;
            }

            const THREE = window.THREE;

            if (!THREE) {
                throw new Error('THREE global not found after loadScript');
            }

            const canvas =
                this.template.querySelector('.scene-canvas');

            this._layer =
                this.template.querySelector('.scene-layer');

            if (!canvas || !this._layer) {
                return;
            }

            this._mobile = this._layer.clientWidth < 820;

            this._ctx = buildScene(THREE, canvas, this._mobile);

            this._resize();

            window.addEventListener('resize', this._onResize);

            if (!this._reduceMotion) {

                window.addEventListener('mousemove', this._onMouseMove, {
                    passive: true
                });

                document.addEventListener(
                    'visibilitychange',
                    this._onVisibility
                );

                if (typeof IntersectionObserver !== 'undefined') {

                    this._pageObserver = new IntersectionObserver(
                        entries => {

                            this._pageVisible = entries[0].isIntersecting;

                            if (this._pageVisible && !document.hidden) {
                                this._start();
                            } else {
                                this._stop();
                            }

                        }
                    );

                    this._pageObserver.observe(page);

                }

                this._start();

            } else {

                window.addEventListener('scroll', this._onScrollStatic, {
                    passive: true,
                    capture: true
                });

                this._frame(0);

            }

            page.classList.add('scene-live');

        } catch (error) {

            // CSS fallback stays visible. The page still works.
            // eslint-disable-next-line no-console
            console.warn('UJJAYINI 3D scene unavailable:', error);

        }

    }


    _resize() {

        const c = this._ctx;

        if (!c || !this._layer) {
            return;
        }

        const width = this._layer.clientWidth || window.innerWidth;
        const height = this._layer.clientHeight || window.innerHeight;

        this._mobile = width < 820;

        c.renderer.setPixelRatio(
            Math.min(window.devicePixelRatio || 1, this._mobile ? 1.5 : 2)
        );

        c.renderer.setSize(width, height, false);

        c.camera.aspect = width / height;

        c.camera.position.z = c.camera.aspect < 0.9 ? 19 : 14;

        c.camera.updateProjectionMatrix();

    }


    _start() {

        if (this._running || this._reduceMotion || !this._ctx) {
            return;
        }

        this._running = true;

        this._rafId = window.requestAnimationFrame(this._tick);

    }


    _stop() {

        this._running = false;

        if (this._rafId) {
            window.cancelAnimationFrame(this._rafId);
            this._rafId = 0;
        }

    }


    _updateProgress() {

        if (!this._page) {
            return;
        }

        const rect = this._page.getBoundingClientRect();

        const total = rect.height - window.innerHeight;

        this._progTarget =
            total > 0
                ? Math.min(1, Math.max(0, -rect.top / total))
                : 0;

    }


    _frame(t) {

        const c = this._ctx;

        if (!c) {
            return;
        }

        this._updateProgress();

        const snap = this._reduceMotion ? 1 : 1;
        const ease = this._reduceMotion ? snap : 0.07;

        this._prog += (this._progTarget - this._prog) * ease;

        const key = sampleKeys(this._prog);

        const targetX = key.x * (this._mobile ? 0.25 : 1);
        const targetY = key.y + (this._mobile ? 1.4 : 0);
        const targetScale = key.s * (this._mobile ? 0.72 : 1);

        const move = this._reduceMotion ? 1 : 0.06;

        this._cur.x += (targetX - this._cur.x) * move;
        this._cur.y += (targetY - this._cur.y) * move;
        this._cur.z += (key.z - this._cur.z) * move;
        this._cur.s += (targetScale - this._cur.s) * move;

        c.planet.position.set(this._cur.x, this._cur.y, this._cur.z);
        c.planet.scale.setScalar(this._cur.s);

        const prog = this._prog;

        c.armillary.rotation.y = t * 0.045 + prog * 1.9;

        c.dial.rotation.z = -t * 0.018 + prog * 0.5;

        c.orbits.forEach(orbit => {
            orbit.pivot.rotation.z = orbit.phase + t * orbit.speed;
        });

        c.farStars.rotation.y = t * 0.003 + prog * 0.25;

        c.nearStars.rotation.y = -t * 0.006 - prog * 0.5;
        c.nearStars.rotation.x = prog * 0.15;

        // Ujjain marker pulse
        c.dotHalo.material.opacity = 0.65 + Math.sin(t * 2.2) * 0.25;

        c.pulses.forEach(pulse => {

            const phase = (t * 0.45 + pulse.offset) % 1;

            const size = 0.3 + phase * 1.6;

            pulse.sprite.scale.set(size, size, 1);

            pulse.sprite.material.opacity = (1 - phase) * 0.8;

        });

        // Mouse parallax
        c.camera.position.x +=
            (this._mouse.x * 0.7 - c.camera.position.x) * 0.04;

        c.camera.position.y +=
            (-this._mouse.y * 0.45 - c.camera.position.y) * 0.04;

        c.camera.lookAt(0, 0, 0);

        c.renderer.render(c.scene, c.camera);

    }


    _disposeScene() {

        this._stop();

        window.removeEventListener('resize', this._onResize);
        window.removeEventListener('mousemove', this._onMouseMove);
        window.removeEventListener('scroll', this._onScrollStatic, true);
        document.removeEventListener('visibilitychange', this._onVisibility);

        if (this._pageObserver) {
            this._pageObserver.disconnect();
            this._pageObserver = null;
        }

        const c = this._ctx;

        if (!c) {
            return;
        }

        c.disposables.forEach(item => {
            if (item && typeof item.dispose === 'function') {
                item.dispose();
            }
        });

        c.renderer.dispose();

        this._ctx = null;

    }


    // =========================================================
    // ACTIONS
    // =========================================================

    handleScrollDown() {

        const target =
            this.template.querySelector('.origin-section');

        if (target) {

            target.scrollIntoView({
                behavior: 'smooth',
                block: 'start'
            });

        }

    }


    handleExplore() {

        window.location.href = '/ujjayini/watches';

    }

}

===== ujjayiniCollectionSection/ujjayiniCollectionSection.js
import { LightningElement, wire } from 'lwc';

import getCollectionsWithWatches
    from '@salesforce/apex/UJJAYINICollectionController.getCollectionsWithWatches';

import HERO_IMAGE
    from '@salesforce/resourceUrl/UJJAYINI_Collections_Hero';

export default class UjjayiniCollectionSection extends LightningElement {

    collections = [];
    errorMessage;


    // =========================================================
    // HERO STATIC RESOURCE
    // =========================================================

    get heroPhotoStyle() {

        return `
            background-image: url("${HERO_IMAGE}");
        `;
    }


    // =========================================================
    // LOAD COLLECTIONS
    // =========================================================

    @wire(getCollectionsWithWatches)
    wiredCollections({ data, error }) {

        if (data) {

            this.collections = data.map(
                (collection, index) => {

                    const displayIndex =
                        String(index + 1).padStart(2, '0');

                    return {
                        ...collection,

                        displayIndex,

                        /*
                         * Apex returns:
                         * description
                         *
                         * HTML expects:
                         * shortDescription
                         */
                        shortDescription:
                            collection.description,

                        categoryLabel:
                            this.getCategoryLabel(
                                collection.name
                            ),

                        signature:
                            this.getSignature(
                                collection.name
                            )
                    };
                }
            );

            this.errorMessage = undefined;

        } else if (error) {

            this.collections = [];

            this.errorMessage =
                this.getErrorMessage(error);
        }
    }


    // =========================================================
    // EMPTY STATE
    // =========================================================

    get showEmpty() {

        return (
            !this.errorMessage &&
            this.collections.length === 0
        );
    }


    // =========================================================
    // COLLECTION CARD CLICK
    // =========================================================
    //
    // ALL collection cards intentionally go to:
    //
    // /ujjayini/watches
    //
    // We DO NOT use:
    //
    // /ujjayini/watches?collection=Ayas
    // /ujjayini/watches?collection=Indranila
    // /ujjayini/watches?collection=Hiranya
    //
    // The Watches page will contain every collection
    // section-wise.
    //
    // =========================================================

    handleCollectionClick() {

        window.location.href =
            '/ujjayini/watches';
    }


    // =========================================================
    // CATEGORY LABEL
    // =========================================================

    getCategoryLabel(name) {

        if (!name) {
            return 'HOROLOGY';
        }

        const normalized =
            name.toLowerCase();

        if (
            normalized.includes('indran')
        ) {
            return 'CELESTIAL';
        }

        if (
            normalized.includes('hiranya')
        ) {
            return 'RADIANCE';
        }

        if (
            normalized.includes('ayas')
        ) {
            return 'PERMANENCE';
        }

        return 'THE COLLECTION';
    }


    // =========================================================
    // SIGNATURE
    // =========================================================

    getSignature(name) {

        if (!name) {
            return 'TIME · UJJAYINI';
        }

        const normalized =
            name.toLowerCase();

        if (
            normalized.includes('indran')
        ) {
            return 'SAPPHIRE · MIDNIGHT';
        }

        if (
            normalized.includes('hiranya')
        ) {
            return 'GOLD · DAWN';
        }

        if (
            normalized.includes('ayas')
        ) {
            return 'STEEL · ETERNAL';
        }

        return 'UJJAYINI · HOROLOGY';
    }


    // =========================================================
    // ERROR MESSAGE
    // =========================================================

    getErrorMessage(error) {

        if (
            error &&
            error.body &&
            error.body.message
        ) {
            return error.body.message;
        }

        if (
            error &&
            error.message
        ) {
            return error.message;
        }

        return (
            'Unable to load the UJJAYINI collections.'
        );
    }
}

function norm(value) {
    return (value || '')
        .toString()
        .normalize('NFD')
        .replace(/[̀-ͯ]/g, '')
        .toLowerCase()
        .trim();
}

// Cloudinary delivers a smaller, modern-format image when
// transformations are added right after "/upload/".
function cloudinary(url, transform) {

    if (!url || url.indexOf('res.cloudinary.com') === -1 || url.indexOf('/upload/') === -1) {
        return url;
    }

    return url.replace('/upload/', '/upload/' + transform + '/');
}

function tagline(description) {

    if (!description) {
        return '';
    }

    const clean = description.replace(/\s+/g, ' ').trim();
    const match = clean.match(/^.*?[.!?](\s|$)/);
    const sentence = (match ? match[0] : clean).trim();
    const line = sentence.length <= 86 ? sentence : sentence.slice(0, 84).replace(/\s+\S*$/, '') + '…';

    return line;
}

            elements.forEach(el => {

                el.classList.add('is-in');

                entries.forEach(entry => {
                    if (entry.isIntersecting) {

                        this._observer.unobserve(entry.target);
                    }
                });

            el.setAttribute('data-watched', 'true');

    get priceBars() {

        const floor = this.priceFloor;
        const span = (this.priceCeil - floor) || 1;
        const counts = new Array(HISTOGRAM_BARS).fill(0);

        this.products.forEach(product => {
            if (product.price > 0) {
                const slot = Math.min(HISTOGRAM_BARS - 1, Math.floor(((product.price - floor) / span) * HISTOGRAM_BARS));
                counts[slot]++;
            }
        });

        const max = Math.max(1, ...counts);

        return counts.map((count, index) => {

            const start = floor + (span / HISTOGRAM_BARS) * index;
            const end = start + span / HISTOGRAM_BARS;
            const inRange = end >= this.priceLow && start <= this.priceHigh;

            return {
                key: 'bar' + index,
                style: 'height:' + (count ? 18 + (count / max) * 82 : 6) + '%;',
                className: inRange ? 'bar is-in' : 'bar'
            };
        });
    }

    // =========================================================
    // FILTERING
    // =========================================================

    passes(product, skipFacet) {

        if (this.activeCollection !== 'ALL' && norm(product.collectionName) !== this.activeCollection) {
            return false;
        }

        if (this.isPriceFiltered && product.price > 0 &&
            (product.price < this.priceLow || product.price > this.priceHigh)) {
            return false;
        }

        if (this.limitedOnly && !product.limitedEdition) {
            return false;
        }

        if (this.inStockOnly && product.stockStatus === 'OUT_OF_STOCK') {
            return false;
        }

        if (this.activeMood) {
            const mood = MOODS.find(m => m.key === this.activeMood);
            if (mood && !mood.test(product)) {
                return false;
            }
        }

        for (const facet of FACETS) {

            if (facet.key === skipFacet) {
                continue;
            }

            const chosen = this.selected[facet.key];

            if (chosen && chosen.length && !chosen.includes(product[facet.key])) {
                return false;
            }
        }

        return true;
    }

    get filteredProducts() {

        const result = this.products.filter(product => this.passes(product));

        result.sort((a, b) => {

            if (this.sortBy === 'price-low') {
                return (a.price || 0) - (b.price || 0);
            }

            if (this.sortBy === 'price-high') {
                return (b.price || 0) - (a.price || 0);
            }

            if (this.sortBy === 'name') {
                return (a.productName || '').localeCompare(b.productName || '');
            }

            return a.order - b.order;
        });

        return result;
    }

    get facetGroups() {

            const counts = new Map();

            this.products.forEach(product => {
                const value = product[facet.key];
                if (value && this.passes(product, facet.key)) {
                    counts.set(value, (counts.get(value) || 0) + 1);
                }
            });

            // Keep chosen values visible even if their count is 0.
            (this.selected[facet.key] || []).forEach(value => {
                if (!counts.has(value)) {
                    counts.set(value, 0);
                }
            });

            const chosen = this.selected[facet.key] || [];

            return {
                ...facet,
                options: [...counts.entries()]
                    .sort((a, b) => a[0].localeCompare(b[0]))
                    .map(([value, count]) => ({
                        key: facet.key + '|' + value,
                        value,
                        count,
                        className: chosen.includes(value) ? 'facet is-on' : 'facet',
                        pressed: chosen.includes(value) ? 'true' : 'false'
                    }))
            };

    handleLoupeMove(event) {

        if (!this.loupeOn || event.pointerType === 'touch') {

        const stage = event.currentTarget;
        const img = stage.querySelector('.watch-image-file');
        const lens = this.template.querySelector('.loupe');

        if (!img || !lens || !img.naturalWidth) {
            return;
        }

        const rect = stage.getBoundingClientRect();
        const x = event.clientX - rect.left;
        const y = event.clientY - rect.top;

        // Geometry of an object-fit: cover image.
        const scale = Math.max(rect.width / img.naturalWidth, rect.height / img.naturalHeight);
        const drawnW = img.naturalWidth * scale;
        const drawnH = img.naturalHeight * scale;
        const offsetX = (rect.width - drawnW) / 2;
        const offsetY = (rect.height - drawnH) / 2;

        const bgW = drawnW * LOUPE_ZOOM;
        const bgH = drawnH * LOUPE_ZOOM;
        const bgX = -((x - offsetX) * LOUPE_ZOOM - LOUPE_RADIUS);
        const bgY = -((y - offsetY) * LOUPE_ZOOM - LOUPE_RADIUS);

        const sharp = stage.dataset.hires || img.currentSrc || img.src;

        lens.style.opacity = '1';
        lens.style.transform = 'translate3d(' + (event.clientX - LOUPE_RADIUS) + 'px,' + (event.clientY - LOUPE_RADIUS) + 'px,0)';
        lens.style.backgroundImage = 'url("' + sharp + '")';
        lens.style.backgroundSize = bgW.toFixed(0) + 'px ' + bgH.toFixed(0) + 'px';
        lens.style.backgroundPosition = bgX.toFixed(1) + 'px ' + bgY.toFixed(1) + 'px';
    }

        let over = false;

        if (dock) {
            const r = dock.getBoundingClientRect();
            over = x > r.left - 30 && x < r.right + 30 && y > r.top - 50 && y < r.bottom + 30;
        }

        if (over !== this.overDock) {
            this.overDock = over;
        }

        this.moveGhost(x, y);

        if (y < 80) {
            window.scrollBy(0, -14);
        } else if (y > window.innerHeight - 140 && !over) {
            window.scrollBy(0, 14);
        }

        if (isGuestUser) {
            this.showGuestCartModal = true;
            return;
        }

        const productId = event.currentTarget.dataset.productId;
        const product = this.products.find(p => p.productId === productId);

        if (!product || product.isAddingToWishlist) {
            return;
        }

        const adding = !product.isWishlisted;

        // Optimistic: the heart reacts instantly.
        this.updateProduct(productId, { isWishlisted: adding, isAddingToWishlist: true });

        if (adding) {
            this.pulse(productId);
        }

        try {

            if (adding) {
                await addToWishlist({ productId });
            } else {
                await removeFromWishlist({ productId });
            }

            this.showNotificationPopup(
                'success',
                adding ? 'SAVED TO WISHLIST' : 'REMOVED FROM WISHLIST',
                adding
                    ? 'The timepiece will wait for you in your wishlist.'
                    : 'The timepiece has left your wishlist.'
            );

        } catch (error) {

            this.updateProduct(productId, { isWishlisted: !adding });
            this.showNotificationPopup('error', 'WISHLIST UNAVAILABLE', this.getErrorMessage(error));

        } finally {

            this.updateProduct(productId, { isAddingToWishlist: false });
        }
