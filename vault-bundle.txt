import { refreshApex } from '@salesforce/apex';

import getMyAssets
    from '@salesforce/apex/UJJAYINICollectorVaultController.getMyAssets';

import verifyPayment
    from '@salesforce/apex/UJJAYINICollectorVaultController.verifyPayment';

import getMyWarranties
    from '@salesforce/apex/UJJAYINIWarrantyController.getMyWarranties';

import vaultBackground
    from '@salesforce/resourceUrl/UJJAYINI_Vault_Teaser_Visual';


// Check every 2 seconds, at most 15 times (about 30 seconds).
const POLL_INTERVAL_MS = 2000;
const MAX_POLL_ATTEMPTS = 15;

const BANNER_COPY = {

    VERIFYING: {
        eyebrow: 'CONFIRMING YOUR PAYMENT',
        title: 'One moment.',
        message:
            'We are verifying your payment with our records.'
    },

    PREPARING: {
        eyebrow: 'PAYMENT RECEIVED',
        title: 'Your timepiece is being prepared.',
        message:
            'It is being registered in your Collector\'s Vault.'
    },

    REGISTERED: {
        eyebrow: 'WELCOME TO YOUR COLLECTION',
        title: 'Your timepiece is now in the Vault.',
        message:
            'Its ownership record is now part of your collection.'
    },

    TIMEOUT_CONFIRMED: {
        eyebrow: 'PAYMENT CONFIRMED',
        title: 'Almost there.',
        message:
            'Your timepiece is still being registered in the Vault. ' +
            'Please refresh shortly.'
    },

    TIMEOUT_UNCONFIRMED: {
        eyebrow: 'PAYMENT UNDER CONFIRMATION',
        title: 'We have not received confirmation yet.',
        message:
            'Your timepiece will appear in the Vault as soon as your ' +

    }

export default class UjjayiniCollectorVault extends LightningElement {

    assets = [];
    warranties = [];

    isLoading = true;
    errorMessage;

    wiredAssetsResult;

    // Post-payment confirmation
    paymentPhase = '';

    _paymentLinkId;
    _paymentConfirmed = false;
    _pollAttempts = 0;
    _pollTimer;
    _dismissTimer;
    _verifyInFlight = false;
    _disposed = false;


    // =========================================================
    // LIFECYCLE
    // =========================================================

    connectedCallback() {

        this.startPaymentVerificationIfNeeded();

    }


    disconnectedCallback() {

        this._disposed = true;

        this.stopPolling();

        if (this._dismissTimer) {
            clearTimeout(this._dismissTimer);
        }

    }


    // =========================================================
    // ASSETS + WARRANTIES (unchanged behaviour)
    // =========================================================

    @wire(getMyAssets)
    wiredAssets(result) {

        this.wiredAssetsResult = result;

        const { data, error } = result;

        if (data) {

            this.assets = data.map(asset => ({
                ...asset
            }));

            this.errorMessage = undefined;

            this.loadWarranties();

        } else if (error) {

            this.isLoading = false;

            this.errorMessage =
                this.getErrorMessage(error);

        }

    }


    async loadWarranties() {

        try {

            const warrantyData =
                await getMyWarranties();

            this.warranties =
                warrantyData || [];

            this.attachWarrantiesToAssets();

        } catch (error) {

            this.errorMessage =
                this.getErrorMessage(error);

        } finally {

            this.isLoading = false;
        }

    }


    attachWarrantiesToAssets() {

        this.assets =
            this.assets.map(asset => {

                const warranty =
                    this.warranties.find(
                        item =>
                            item.assetId === asset.assetId
                    );

                return {
                    ...asset,
                    warranty: warranty || null
                };

            });

    }


    get vaultBackgroundStyle() {

        return `background-image: url("${vaultBackground}");`;

    }


    get hasAssets() {

        return this.assets.length > 0;

    }


    get hasMultipleAssets() {

        return this.assets.length > 1;

    }


    get isEmpty() {

        return (
            !this.isLoading &&
            !this.errorMessage &&
            !this.isVerifyingPayment &&
            this.assets.length === 0
        );

    }


    // =========================================================
    // POST-PAYMENT VERIFICATION
    //
    // The URL parameter is only a trigger. Salesforce decides
    // whether the payment is captured and the Asset exists.
    // =========================================================

    startPaymentVerificationIfNeeded() {

        let params;

        try {

            params = new URLSearchParams(window.location.search);

        } catch (error) {

            return;

        }

        const status =
            params.get('razorpay_payment_link_status');

        if (!status || status.toLowerCase() !== 'paid') {
            return;
        }

        this._paymentLinkId =
            params.get('razorpay_payment_link_id');

        this._paymentConfirmed = false;

        this._pollAttempts = 0;

        this.paymentPhase = 'VERIFYING';

        this.verifyPaymentOnce();

    }


    async verifyPaymentOnce() {

        if (this._disposed || this._verifyInFlight) {
            return;
        }

        this._verifyInFlight = true;

        this._pollAttempts += 1;

        let verification = null;

        try {

            verification = await verifyPayment({
                paymentLinkId: this._paymentLinkId
            });

        } catch (error) {

            // A network hiccup must not stop the polling.
            verification = null;

        } finally {

            this._verifyInFlight = false;

        }

        if (this._disposed) {
            return;
        }

        if (verification && verification.assetReady) {

            await this.finishPaymentVerification();

            return;
        }

        if (verification && verification.paymentConfirmed) {
            this._paymentConfirmed = true;
        }

        if (this._pollAttempts >= MAX_POLL_ATTEMPTS) {

            this.paymentPhase =
                this._paymentConfirmed
                    ? 'TIMEOUT_CONFIRMED'
                    : 'TIMEOUT_UNCONFIRMED';

            return;
        }

        this.paymentPhase =
            this._paymentConfirmed
                ? 'PREPARING'
                : 'VERIFYING';

        this._pollTimer = setTimeout(
            () => this.verifyPaymentOnce(),
            POLL_INTERVAL_MS
        );

    }


    async finishPaymentVerification() {

        this.stopPolling();

        try {

            if (this.wiredAssetsResult) {
                await refreshApex(this.wiredAssetsResult);
            }

        } catch (error) {

            // The wire will deliver the data on its next refresh.

        }

        if (this._disposed) {
            return;
        }

        this.paymentPhase = 'REGISTERED';

        this.removePaymentParamsFromUrl();

        this._dismissTimer = setTimeout(() => {

            if (this.paymentPhase === 'REGISTERED') {
                this.paymentPhase = '';
            }

        }, 12000);

    }


    stopPolling() {

        if (this._pollTimer) {

            clearTimeout(this._pollTimer);

            this._pollTimer = null;
        }

    }


    removePaymentParamsFromUrl() {

        try {

            const url = new URL(window.location.href);

            const razorpayKeys =
                Array.from(url.searchParams.keys())
                    .filter(key => key.startsWith('razorpay_'));

            if (razorpayKeys.length === 0) {
                return;
            }

            razorpayKeys.forEach(key => {
                url.searchParams.delete(key);
            });

            window.history.replaceState(
                window.history.state,
                '',
                url.pathname + url.search + url.hash
            );

        } catch (error) {

            // Cleaning the address bar is cosmetic only.

        }

    }


    handleDismissPayment() {

        this.paymentPhase = '';

    }


    handleRefreshVault() {

        window.location.reload();

    }


    // =========================================================
    // PAYMENT BANNER
    // =========================================================

    get showPaymentBanner() {

        return !!this.paymentPhase;

    }


    get isVerifyingPayment() {

        return (
            this.paymentPhase === 'VERIFYING' ||
            this.paymentPhase === 'PREPARING'
        );

    }


    get showRefresh() {

        return (
            this.paymentPhase === 'TIMEOUT_CONFIRMED' ||
            this.paymentPhase === 'TIMEOUT_UNCONFIRMED'
        );

    }


    get paymentBannerClass() {

        return this.isVerifyingPayment
            ? 'payment-banner is-working'
            : 'payment-banner is-done';

    }


    get paymentBannerEyebrow() {

        return (BANNER_COPY[this.paymentPhase] || {}).eyebrow;

    }


    get paymentBannerTitle() {

        return (BANNER_COPY[this.paymentPhase] || {}).title;

    }


    get paymentBannerMessage() {

        return (BANNER_COPY[this.paymentPhase] || {}).message;

    }


    // =========================================================
    // ERROR
    // =========================================================

    getErrorMessage(error) {

        if (
            error &&
            error.body &&
            error.body.message
        ) {

            return error.body.message;

        }


        if (error?.message) {

            return error.message;

        }


        return 'Unable to load your collection.';

    }

}

    @AuraEnabled
    public static PaymentVerificationResult verifyPayment(
        String paymentLinkId
    ) {

        PaymentVerificationResult result =
            new PaymentVerificationResult();

        result.state = 'NOT_FOUND';
        result.paymentConfirmed = false;
        result.assetReady = false;

        Id accountId =
            getCurrentAccountId();

        if (accountId == null) {
            return result;
        }

        Payment__c payment =
            new PaymentLookup().findOwnedPayment(
                accountId,
                paymentLinkId
            );

        if (payment == null) {
            return result;
        }

        if (payment.Payment_Status__c != 'Captured') {

            result.state = 'PENDING';

            return result;
        }

        result.paymentConfirmed = true;
        result.state = 'PAYMENT_CONFIRMED';

        /*
         * AssetFulfillmentService marks every Asset of an Order
         * with the serial marker UJJAYINI-<OrderId>.
         */
        String fulfillmentMarker =
            'UJJAYINI-' + String.valueOf(payment.Order__c);

        Integer assetCount = [
            SELECT COUNT()
            FROM Asset
            WHERE AccountId = :accountId
            AND SerialNumber = :fulfillmentMarker
            WITH SYSTEM_MODE
        ];

        if (assetCount > 0) {

            result.assetReady = true;
            result.state = 'ASSET_READY';
        }

        return result;
    }

                   Is_Cancelled__c,

                   ShippingCountry,

            LIMIT 1

            throw new IllegalArgumentException('Order not found.');

                   UnitPrice,

            WITH SYSTEM_MODE
            ORDER BY CreatedDate ASC
        ];

    private static Payment__c latestPayment(Id orderId) {

        List<Payment__c> payments = [
            SELECT Id,
                   Name,
                   Amount__c,
                   Payment_Status__c,
                   Gateway_Payment_ID__c,
                   Paid_At__c,
                   Refunded_At__c
            FROM Payment__c
            WHERE Order__c = :orderId
            WITH SYSTEM_MODE
            ORDER BY Paid_At__c DESC NULLS LAST, CreatedDate DESC
            LIMIT 1
        ];

        return payments.isEmpty() ? null : payments[0];
    }


    private static Contact findCustomer(Order orderRecord) {

        Id contactId =
            orderRecord.BillToContactId != null
                ? orderRecord.BillToContactId
                : orderRecord.ShipToContactId;

        if (contactId != null) {

            List<Contact> contacts = [
                SELECT Id, Name, Email, Phone
                FROM Contact
                WHERE Id = :contactId
                WITH SYSTEM_MODE
                LIMIT 1
            ];

            if (!contacts.isEmpty()) {
                return contacts[0];
            }
        }

        List<Contact> fallback = [
            SELECT Id, Name, Email, Phone
            FROM Contact
            WHERE AccountId = :orderRecord.AccountId
            AND Email != null
            WITH SYSTEM_MODE
            ORDER BY CreatedDate ASC
            LIMIT 1
        ];

        return fallback.isEmpty() ? null : fallback[0];
    }

    public static Boolean sendInvoiceEmail(Id orderId) {

    private static Date invoiceDate(Order orderRecord) {

        if (orderRecord.ActivatedDate != null) {
            return orderRecord.ActivatedDate.date();
        }

        return orderRecord.EffectiveDate != null
            ? orderRecord.EffectiveDate
            : Date.today();
    }


    private static String addressHtml(
        String street,
        String city,
        String state,
        String postalCode,
        String country
    ) {

        if (String.isBlank(street) && String.isBlank(city)) {
            return '<div class="muted">Address not recorded</div>';
        }

        List<String> lines = new List<String>();

        if (String.isNotBlank(street)) {
            for (String line : street.replace('\r\n', '\n').split('\n')) {
                if (String.isNotBlank(line)) {
                    lines.add(esc(line.trim()));
                }
            }
        }

        List<String> cityParts = new List<String>();

        if (String.isNotBlank(city)) { cityParts.add(city); }
        if (String.isNotBlank(state)) { cityParts.add(state); }
        if (String.isNotBlank(postalCode)) { cityParts.add(postalCode); }

        if (!cityParts.isEmpty()) {
            lines.add(esc(String.join(cityParts, ', ')));
        }

        if (String.isNotBlank(country)) {
            lines.add(esc(country));
        }

        return '<div class="muted">' + String.join(lines, '<br/>') + '</div>';
    }


    private static String formatDate(Date value) {

        if (value == null) {
            return '';
        }

        return Datetime.newInstance(
            value,
            Time.newInstance(12, 0, 0, 0)
        ).format('dd MMM yyyy');
    }


      @TestVisible
    private static String formatInr(Decimal amount) {

        Decimal value =
            amount == null
                ? 0
                : amount.setScale(2, RoundingMode.HALF_UP);

        Boolean negative = value < 0;

        String plain = value.abs().toPlainString();

        String whole = plain.contains('.') ? plain.substringBefore('.') : plain;
        String fraction = plain.contains('.') ? plain.substringAfter('.') : '00';

        String grouped = whole;

        if (whole.length() > 3) {

            // Indian grouping: last three digits, then pairs.
            // 165000 -> 1,65,000    1234567 -> 12,34,567
            String lastThree = whole.right(3);
            String rest = whole.left(whole.length() - 3);
            String pairs = '';

            while (rest.length() > 2) {
                pairs = ',' + rest.right(2) + pairs;
                rest = rest.left(rest.length() - 2);
            }

            grouped = rest + pairs + ',' + lastThree;
        }

        return (negative ? '-' : '') +
            'INR ' + grouped + '.' + fraction.rightPad(2, '0').left(2);
    }


    private static String esc(String value) {

        return value == null
            ? ''
            : value.escapeHtml4();
    }
